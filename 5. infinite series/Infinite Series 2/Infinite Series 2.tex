\section*{Problem 1}
Assume $\sin(z) = 0$. Then we have
$$\sin(z) = \frac{e^{iz} - e^{-iz}}{2i} = 0$$
$$e^{iz} = e^{-iz}, \quad \implies \quad e^{2iz} = 1.$$
If $e^w = 1$, then $w \in 2\pi i\mathbb{Z}$. So we have that $2iz = 2\pi in$ for some $n \in \mathbb{Z}$, which gives us $z = \pi n$.

\section*{Problem 2}
We can write $\zeta(k)$ as
$$\zeta(k) = \summation[1] \frac{1}{n^k} = \sum_{\text{odd } n} \frac{1}{n^k} + \sum_{\text{even } n} \frac{1}{n^k} = \sum_{\text{odd } n} \frac{1}{n^k} + \frac{1}{2^k}\zeta(k).$$
So we can write the sum as
$$\summation[1] \frac{1}{(2n - 1)^k} = \sum_{\text{odd } n} \frac{1}{n^k} = \zeta(k) - \frac{1}{2^k}\zeta(k) = \zeta(k)(1 - 2^{-k}).$$
Similarly,
$$\zeta_a(k) = \summation[1] \frac{(-1)^{n - 1}}{n^k} = \sum_{\text{odd } n} \frac{1}{n^k} - \sum_{\text{even } n} \frac{1}{n^k},$$
$$= \summation[1] \frac{1}{(2n - 1)^k} - \frac{1}{n^k}\zeta(k) = (1 - 2^{-k})\zeta(k) - 2^{-k}\zeta(k) = \zeta(k)(1 - 2^{1 - k}).$$
So
$$\summation[1] \frac{1}{(2n - 1)^k} = \frac{1}{1 - 2^{1 - k}}\zeta_a(k).$$

\section*{Problem 5}
\subsection*{Part (a)}
First, we compute the inner integral by setting $u = xy$, so $du = y\,xy$ and $dx = du/y$.
$$I = \int_0^1 \frac{dx}{1 + (xy)^2} = \frac{1}{y}\int_0^y \frac{du}{1 + u^2} = \frac{\arctan y}{y}.$$
Then, we substitute the power series of $\arctan u$:
$$\frac{\arctan y}{y} = \summation \frac{(-1)^n y^{2n}}{2n + 1}.$$
Interchanging the sum and integral, we get
$$I = \summation \frac{(-1)^n}{2n + 1}\int_0^1 y^{2n}dy = \summation \frac{(-1)^n}{(2n + 1)^2} = G.$$

\section*{Problem 9}
\subsection*{Part (a)}
$$\frac{1}{m^n} + \frac{1}{m^{2n}} + \frac{1}{m^{3n}} \cdots = \frac{1/m^n}{1 - 1/m^n} = \frac{1}{m^n - 1}.$$

\subsection*{Part (c)}
Swap the order:
$$\summationk[2]\summation[2] \frac{1}{n^k} = \summation[2]\summationk[2] \frac{1}{n^k}.$$
Now we evaluate the inner sum:
$$\summationk[2] \frac{1}{n^k} = \frac{1/n^2}{1 - 1/n} = \frac{1}{n(n - 1)}.$$
So
$$\summation[2]\summationk[2] \frac{1}{n^k} = \summation[2] \frac{1}{n(n - 1)} = \summation[2] \bigp{\frac{1}{n - 1} - \frac{1}{n}} = 1.$$

\section*{Problem 10}
The denominator $m^n - 1$ is odd exactly when $m$ is even. So we can evaluate the same sum from problem 9 but restrict $m$ to even values:
$$\sum_{n \text{ even } \ge 2}\summationk[2] \frac{1}{n^k} = \sum_{n \text{ even } \ge 2} \frac{1}{n(n - 1)} = \summationm[1] \frac{1}{(2m)(2m - 1)}$$
$$= \summationm[1] \bigp{\frac{1}{2m - 1} - \frac{1}{2m}} = 1 - \frac{1}{2} + \frac{1}{3} - \cdots = \ln 2.$$

\section*{Problem 13}
By Theorem 2.1,
$$\pi\cot \pi x = \frac{1}{x} + \summation[1]\frac{2x}{x^2 - n^2}$$
for all $x \in \mathbb{C} \setminus \mathbb{Z}$.
Plugging in $x = i$, we get
$$\pi \cot (i\pi) = -i - 2i\summation \frac{1}{n^2 + 1},$$
$$\summation \frac{1}{n^2 + 1} = - \frac{\pi\cot(i\pi) + i}{2i} = -\frac{-i\pi\coth \pi + i}{2i} = \frac{\pi\coth\pi - 1}{2}.$$
